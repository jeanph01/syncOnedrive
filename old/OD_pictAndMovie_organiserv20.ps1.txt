param (
    [bool]$Execute = $false,
    [string]$IndexFile = ".\onedrive_cache.json",
    [string]$LogFile = ".\organisation_log.txt",
    [string]$ProcessedLog = ".\processed_ids.log",
    [string]$ExecutionReport = ".\azure_sync_report.csv",
    [bool]$VerboseMode = $false
)

# --- CONFIGURATION GLOBALE ---
$ExtensionMap = @{}
# On définit les groupes
@(".jpg", ".jpeg", ".png", ".heic", ".bmp", ".gif", ".pcx") | ForEach-Object { $ExtensionMap[$_] = "Images" }
@(".mp4", ".mov", ".avi", ".mpg", ".mpeg", ".wmv")          | ForEach-Object { $ExtensionMap[$_] = "Videos" }
@(".mp3", ".m4a")                                           | ForEach-Object { $ExtensionMap[$_] = "Audio" }

$Config = [PSCustomObject]@{
    RenameMarker = "_v_"
    MaxPathLen   = 100
    ClientId     = "176fc7bc-42c9-4a25-82b5-0ad584d3c061"
    ExtensionMap = $ExtensionMap
}

# État du script (évite les variables globales éparpillées)
$Global:State = @{
    Headers        = $null
    PlannedActions = New-Object System.Collections.Generic.List[PSCustomObject]
    Cache          = $null
    ProcessedIds   = @{}
}

# --- FONCTIONS UTILITAIRES ---

function Write-Log {
    param([string]$Message, [string]$Level = "INFO")
    $Timestamp = Get-Date -Format "HH:mm:ss"
    $Color = switch($Level) {
        "ERROR"   { "Red" }
        "WARN"    { "Yellow" }
        "SUCCESS" { "Green" }
        "DEBUG"   { "DarkGray" }
        default   { "Gray" }
    }
    if ($Level -ne "DEBUG" -or $VerboseMode) {
        Write-Host "[$Timestamp] [$Level] $Message" -ForegroundColor $Color
    }
    "[$Timestamp] [$Level] $Message" | Add-Content $LogFile -ErrorAction SilentlyContinue
}

function Get-ErrorDetails {
    param($Ex)
    try {
        if ($Ex.Response) {
            $reader = New-Object System.IO.StreamReader($Ex.Response.GetResponseStream())
            return $reader.ReadToEnd()
        }
    } catch {}
    return $Ex.Message
}

function Normalize-ToAscii {
    param([string]$text, [bool]$isPath = $false)
    if ([string]::IsNullOrWhiteSpace($text)) { return "" }
    $clean = $text.Normalize([System.Text.NormalizationForm]::FormD) -replace '\p{M}', ''
    $clean = $clean -replace "[a-fA-F0-9]{8}-([a-fA-F0-9]{4}-){3}[a-fA-F0-9]{12}", ""
    $clean = $clean -replace "[a-zA-Z0-9]{16,}", ""
    $pattern = if ($isPath) { "[^a-zA-Z0-9\.\-/]" } else { "[^a-zA-Z0-9\.\-]" }
    return ($clean -replace $pattern, "_" -replace "_+", "_").Trim("_")
}

# --- LOGIQUE MÉTIER ---

function Build-SmartFileName {
    param($timestampStr, $contextStr, $originalName, $extension)
    
    $cleanOriginal = Normalize-ToAscii $originalName
    $cleanContext  = Normalize-ToAscii $contextStr
    $filteredWords = ($cleanOriginal -split "_" | Where-Object { $_.Length -gt 1 -and $timestampStr -notmatch $_ }) -join "_"

    $fixedLen = $timestampStr.Length + $filteredWords.Length + $Config.RenameMarker.Length + $extension.Length + 5
    $availableForContext = $Config.MaxPathLen - $fixedLen
    
    $finalContext = ""
    if ($availableForContext -gt 5 -and $cleanContext) {
        $finalContext = if ($cleanContext.Length -gt $availableForContext) { $cleanContext.Substring($cleanContext.Length - $availableForContext) } else { $cleanContext }
    }

    $parts = @($timestampStr, $filteredWords, $finalContext) | Where-Object { $_ }
    return (($parts -join "_").Trim("_") + $Config.RenameMarker + $extension)
}

function Determine-TargetRoot {
    param($SourcePath)
    if ($SourcePath -like "*Pour coffre fort/Michelle*") { return "Pour coffre fort/Michelle" }
    if ($SourcePath -like "*Pour coffre fort/relations*") { return "Pour coffre fort/relations" }
    if ($SourcePath -like "*Pour coffre fort/archives*") { return "Pour coffre fort/archives" }
    return if ($SourcePath -match "Videos") { "Videos" } else { "Images/Pellicule" }
}

# --- ACTIONS ONEDRIVE ---

function Authenticate-Azure {
    Write-Log "Demande de code de connexion..." "WARN"
    $DeviceCodeRequest = Invoke-RestMethod -Method POST -Uri "https://login.microsoftonline.com/common/oauth2/v2.0/devicecode" -Body @{ client_id = $Config.ClientId; scope = "Files.ReadWrite.All" }
    
    Write-Host "`n--- ACTION REQUISE ---" -ForegroundColor Cyan
    Write-Host "Connectez-vous ici : https://microsoft.com/devicelogin"
    Write-Host "Code : $($DeviceCodeRequest.user_code)" -ForegroundColor Yellow -Object
    Write-Host "----------------------`n"

    while (!$Global:State.Headers) {
        Start-Sleep 5
        try {
            $Auth = Invoke-RestMethod -Method POST -Uri "https://login.microsoftonline.com/common/oauth2/v2.0/token" -Body @{ 
                grant_type = "urn:ietf:params:oauth:grant-type:device_code"
                client_id  = $Config.ClientId
                device_code = $DeviceCodeRequest.device_code 
            }
            $Global:State.Headers = @{ Authorization = "Bearer $($Auth.access_token)"; "Content-Type" = "application/json" }
        } catch { }
    }
    Write-Log "Authentification réussie." "SUCCESS"
}

function Ensure-OneDrivePath {
    param($Path)
    $Path = $Path.Trim('/')
    $parts = $Path -split '/'
    $currentPath = ""
    foreach ($part in $parts) {
        $parentPath = $currentPath
        $currentPath += if ($currentPath -eq "") { $part } else { "/$part" }
        $segments = ($currentPath -split '/' | ForEach-Object { [Uri]::EscapeDataString($_) }) -join '/'
        
        try {
            Invoke-RestMethod -Headers $Global:State.Headers -Uri "https://graph.microsoft.com/v1.0/me/drive/root:/$($segments):" -Method Get -ErrorAction Stop > $null
        } catch {
            Write-Log "Création du dossier : /$currentPath" "DEBUG"
            $parentSegments = ($parentPath -split '/' | Where-Object {$_} | ForEach-Object { [Uri]::EscapeDataString($_) }) -join '/'
            $uriPost = if ($parentPath -eq "") { "https://graph.microsoft.com/v1.0/me/drive/root/children" } else { "https://graph.microsoft.com/v1.0/me/drive/root:/$($parentSegments):/children" }
            $body = @{ name = $part; folder = @{}; "@microsoft.graph.conflictBehavior" = "ignore" } | ConvertTo-Json
            Invoke-RestMethod -Headers $Global:State.Headers -Uri $uriPost -Method POST -Body $body -ErrorAction Stop > $null
        }
    }
}

# --- CORE ---

function Main {
    Clear-Host
    Write-Log "Démarrage de l'organisateur..."
    
    # 1. Init & Cache
    if (Test-Path $LogFile) { Remove-Item $LogFile }
    "Timestamp,ID,Status,OldPath,NewPath,Error" | Set-Content $ExecutionReport
    
    if (!(Test-Path $IndexFile)) { Write-Log "Cache introuvable !" "ERROR"; return }
    $Global:State.Cache = Get-Content $IndexFile -Raw | ConvertFrom-Json -AsHashtable
    if (Test-Path $ProcessedLog) { Get-Content $ProcessedLog | ForEach-Object { $Global:State.ProcessedIds[$_] = $true } }

    # 2. Planification
    Write-Log "Analyse des fichiers en cours..."
    $FolderInventory = @{}

    foreach ($fileId in $Global:State.Cache.Files.Keys) {
        $file = $Global:State.Cache.Files[$fileId]
        $ext = [System.IO.Path]::GetExtension($file.n).ToLower()
        
        # Filtrage
        if ($Global:State.ProcessedIds.ContainsKey($fileId) -or $file.n -match $Config.RenameMarker) { continue }
        
        $subType = "Autres"
        foreach($key in $Config.ExtensionMap.Keys) { if ($key -contains $ext) { $subType = $Config.ExtensionMap[$key]; break } }
        if ($subType -eq "Autres") { continue }

        # Calcul Destination
        $date = [DateTime]$file.d
        $targetRoot = Determine-TargetRoot $file.p
        $rawDest = if ($targetRoot -like "Pour coffre fort*") { "$targetRoot/$subType/$($date.Year)/$($date.ToString('MM'))" } else { "$targetRoot/$($date.Year)/$($date.ToString('MM'))" }
        $cleanDest = Normalize-ToAscii $rawDest -isPath $true
        
        # Nom intelligent
        $pathParts = $file.p -split "/" | Where-Object { $_ -and $_ -notmatch "drive|root|Images|Videos|Pellicule" }
        $context = if ($pathParts.Count -gt 0) { ($pathParts[[Math]::Max(0, $pathParts.Count-3)..($pathParts.Count-1)]) -join "_" } else { "" }
        $newName = Build-SmartFileName $date.ToString("yyyyMMdd_HHmmss") $context ([System.IO.Path]::GetFileNameWithoutExtension($file.n)) $ext

        $targetKey = "/$cleanDest/$newName"
        $Global:State.PlannedActions.Add([PSCustomObject]@{ Id=$fileId; Src=$file.p; Name=$file.n; DstDir=$cleanDest; DstName=$newName; Full=$targetKey })
    }

    Write-Log "Plan généré : $($Global:State.PlannedActions.Count) fichiers à traiter." "SUCCESS"
    if ($Global:State.PlannedActions.Count -eq 0) { return }

    # 3. Exécution
    if ($Execute) {
        Authenticate-Azure
        
        # Pré-création dossiers
        $dirs = $Global:State.PlannedActions.DstDir | Select-Object -Unique | Sort-Object
        foreach ($d in $dirs) { Ensure-OneDrivePath $d }

        # Déplacement
        $count = 0
        foreach ($act in $Global:State.PlannedActions) {
            $count++
            Write-Progress -Activity "Déplacement OneDrive" -Status "Fichier $count / $($Global:State.PlannedActions.Count)" -PercentComplete (($count / $Global:State.PlannedActions.Count) * 100)
            
            Write-Log "Traitement : $($act.DstName)" "DEBUG"
            try {
                $body = @{ parentReference = @{ path = "/drive/root:/$($act.DstDir)" }; name = $act.DstName } | ConvertTo-Json
                Invoke-RestMethod -Headers $Global:State.Headers -Uri "https://graph.microsoft.com/v1.0/me/drive/items/$($act.Id)" -Method PATCH -Body $body -ErrorAction Stop > $null
                
                "$(Get-Date -Format 'HH:mm'),$($act.Id),SUCCESS,$($act.Src),$($act.Full)," | Add-Content $ExecutionReport
                $act.Id | Add-Content $ProcessedLog
                
                # Update Cache local
                $Global:State.Cache.Files[$act.Id].p = "/drive/root:/$($act.DstDir)"
                $Global:State.Cache.Files[$act.Id].n = $act.DstName
            } catch {
                $err = Get-ErrorDetails $_
                Write-Log "Erreur sur $($act.Id) : $err" "ERROR"
                "$(Get-Date -Format 'HH:mm'),$($act.Id),ERROR,$($act.Src),$($act.Full),$err" | Add-Content $ExecutionReport
            }
        }
        $Global:State.Cache | ConvertTo-Json -Depth 10 | Set-Content $IndexFile
        Write-Log "Traitement terminé." "SUCCESS"
    } else {
        Write-Log "MODE APERÇU : Utilisez -Execute `$true pour appliquer les changements." "WARN"
        if ($Global:State.PlannedActions.Count -gt 0) { 
            Write-Host "`nExemple de modification :" -ForegroundColor Cyan
            $Global:State.PlannedActions[0] | Format-List
        }
    }
}

Main
